export const notes = [{id:'n1',owner:'alice',title:'Old',body:'Keep me'}];
export function renameNote(actor,id,title){const note=notes.find(n=>n.id===id);if(!note) throw new Error('missing');if(note.owner!==actor) throw new Error('forbidden');const trimmed=title.trim();if(!trimmed) throw new Error('empty title');note.title=trimmed;return {...note};}
